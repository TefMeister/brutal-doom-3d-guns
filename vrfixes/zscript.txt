version "4.10"
// VR fixes for Brutal Doom with our 3D guns (brutal-doom-3d-guns, 2026-10-19).
//
// Tracers from the gun hand (Tefa: "bullets seem to fly out of the bottom of the screen, about 10-20cm below the gun
// for anything other than the pistol"). Brutal Doom fires its visible tracer (DecorativeTracer) with a spawn height of
// -12 for the shotgun and the rifle, which on a flat screen lines it up with the gun picture at the bottom of the view;
// the pistol uses 0. In VR the spawn point is the controller (AttackPos), so -12 puts the tracer about 35 cm under the
// gun (12 map units at 34 a metre). Each tracer the player fires is lifted to the gun hand's height the moment it
// spawns, so every gun matches the pistol. Flat play (no VR attack position) is left alone.
// The minigun fires the same tracer at the same -12, so it is covered too.
//
// Spent shells (Tefa, the same day: "please fix them too"): every gun throws its casings with an invisible spawner
// fired 9 to 26 units low for the same flat-screen reason. The spawners are lifted the same way before they act.
//
// Every gun, checked (2026-10-19, Tefa: "go over all of them to see if others have the same fault"): plasma balls,
// rockets (rocket, nuke, incendiary), Devastator missiles, flamethrower / flame cannon fire and dropped clips are fired
// 3 to 12 units low too. So the rule is general: anything the player fires as a missile from beside the gun hand starts
// at the gun hand's height. Melee "missiles" (kicks, the axe, the chainsaw, punches, grabs) keep their heights on purpose.

class BDVRFixesHandler : EventHandler
{
	int logged;


	override void WorldThingSpawned(WorldEvent e)
	{
		static const Name spawners[] = { 'PistolCaseSpawn', '50CaseSpawn2', 'ShotCaseSpawn', 'RifleCaseSpawn',
			'PlasmaCaseSpawn', 'SSGCaseSpawner', 'RocketCaseSpawn', 'GrenadeCasingSpawner', 'EmptyASGAmmoSpawn',
			'EmptyClip2Spawn', 'EmptyClipSpawn', 'EmptySniperAmmoSpawn' };
		static const String melee[] = { "Attack", "Kick", "Strike", "Melee", "Saw", "Pick", "Grab", "LostSoul", "LSOHK" };
		let t = e.Thing;
		if (!t) return;
		Name cls = t.GetClassName();
		String scls = cls;
		for (int i = 0; i < melee.Size(); i++) if (scls.IndexOf(melee[i]) >= 0) return;
		bool ours = t.bMISSILE;
		for (int i = 0; i < spawners.Size() && !ours; i++) if (cls == spawners[i]) ours = true;
		if (!ours) return;
		let pmo = players[consoleplayer].mo;
		if (!pmo || t.target != pmo || !pmo.OverrideAttackPosDir) return;
		double dz = pmo.AttackPos.Z - t.pos.Z;
		if (abs(dz) > 48) return;      // not from the gun hand: leave it
		if ((t.pos.XY - pmo.AttackPos.XY).Length() > 64) return;   // fired from somewhere else (a turret, a vehicle)
		if (logged < 12)
		{
			logged++;
			Console.Printf("bd3d vr fix: %s from %s lifted %.1f units to the gun hand", cls,
				pmo.player.ReadyWeapon ? pmo.player.ReadyWeapon.GetClassName() : 'none', dz);
		}
		t.SetOrigin((t.pos.X, t.pos.Y, pmo.AttackPos.Z), false);
	}
}
